#include #include #define IR_SEND_PIN 3 // LED IR sur la pin D3 #define OSRAM_ADDR 0x00 // Adresse télécommande Osram void setup() { Serial.begin(9600); IrSender.begin(IR_SEND_PIN); Serial.println("Arduino IR Osram 24 Touches Pret !"); } void sendCmd(uint8_t command) { IrSender.sendNEC(OSRAM_ADDR, command, 0); Serial.print("Commande IR envoyee : 0x"); Serial.println(command, HEX); } void loop() { if (Serial.available() > 0) { String cmd = Serial.readStringUntil('\n'); cmd.trim(); cmd.toUpperCase(); // Accepte majuscules et minuscules // Ligne 1 : Contrôle & Alimentation if (cmd == "UP" || cmd == "BRIGHTNESS_UP" || cmd == "LUM+" || cmd == "+") sendCmd(0x00); else if (cmd == "DOWN" || cmd == "BRIGHTNESS_DOWN" || cmd == "LUM-" || cmd == "-") sendCmd(0x02); else if (cmd == "POWER" || cmd == "ON") sendCmd(0x07); else if (cmd == "OFF") sendCmd(0x06); // Ligne 2 : Couleurs principales else if (cmd == "R" || cmd == "RED") sendCmd(0x08); else if (cmd == "G" || cmd == "GREEN") sendCmd(0x04); else if (cmd == "B" || cmd == "BLUE") sendCmd(0x0A); else if (cmd == "WHITE" || cmd == "W") sendCmd(0x05); // Ligne 3 à 6 : Nuances Rouges / Oranges else if (cmd == "R1") sendCmd(0x09); else if (cmd == "R2") sendCmd(0x0D); else if (cmd == "R3") sendCmd(0x11); else if (cmd == "R4") sendCmd(0x15); // Ligne 3 à 6 : Nuances Vertes / Cyans else if (cmd == "G1") sendCmd(0x0E); else if (cmd == "G2") sendCmd(0x12); else if (cmd == "G3") sendCmd(0x16); else if (cmd == "G4") sendCmd(0x1A); // Ligne 3 à 6 : Nuances Bleues / Violettes else if (cmd == "B1") sendCmd(0x10); else if (cmd == "B2") sendCmd(0x14); else if (cmd == "B3") sendCmd(0x18); else if (cmd == "B4") sendCmd(0x1C); // Effets spéciaux else if (cmd == "FLASH") sendCmd(0x0F); else if (cmd == "STROBE") sendCmd(0x13); else if (cmd == "SMOOTH") sendCmd(0x17); else if (cmd == "MODE") sendCmd(0x0B); } }
